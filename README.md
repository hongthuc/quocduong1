    function formatCurrencyVND(amount) {
      return amount.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".") + "₫";
    }

    /**
     * Render danh sách sản phẩm ra lưới (Chia 2 cột trên điện thoại, có giá giảm & % giảm giá giống Flatsome, không để nhãn dán đè lên ảnh)
     */
    function renderProducts(filterCategory = "all", searchKeyword = "") {
      const grid = document.getElementById("product-grid");
      grid.innerHTML = "";

      const normalizedSearch = searchKeyword.trim().toLowerCase();
      const filtered = PRODUCTS.filter(item => {
        const matchCat = filterCategory === "all" || item.category === filterCategory;
        const matchSearch = !normalizedSearch ||
          item.name.toLowerCase().includes(normalizedSearch) ||
          item.categoryName.toLowerCase().includes(normalizedSearch);
        return matchCat && matchSearch;
      });

      if (filtered.length === 0) {
        grid.innerHTML = `
          <div class="col-span-full bg-white rounded-xl p-8 text-center border border-gray-200">
            <p class="text-sm text-gray-600 font-semibold">Không tìm thấy sản phẩm phù hợp với từ khóa của bạn.</p>
            <button type="button" onclick="filterByCategory('all')" class="mt-3 px-4 py-2 rounded-lg bg-qd-primary text-white text-xs font-bold">
              Xem tất cả sản phẩm
            </button>
          </div>
        `;
        return;
      }

      filtered.forEach(product => {
        const card = document.createElement("div");
        card.className = "product-small box";

        card.innerHTML = `
          <div class="box-image cursor-pointer" onclick="openProductDetail(${product.id})">
            <img alt="${product.name}" data-product-img="${product.driveId}">
          </div>
          <div class="box-text p-2.5 sm:p-3.5 flex-1 flex flex-col justify-between text-center sm:text-left">
            <div>
              <p class="text-[10px] sm:text-[11px] font-semibold text-gray-400 uppercase truncate">
                ${product.categoryName}
              </p>
              <h3 onclick="openProductDetail(${product.id})" class="font-display font-bold text-xs sm:text-sm text-gray-800 hover:text-qd-primary cursor-pointer line-clamp-2 mt-0.5 min-h-[2.4rem] leading-snug">
                ${product.name}
              </h3>
            </div>
            <div class="mt-2 pt-2 border-t border-gray-100">
              <div class="price-wrapper flex flex-wrap items-center justify-center sm:justify-start gap-y-1">
                <del>${formatCurrencyVND(product.oldPrice)}</del>
                <ins>${formatCurrencyVND(product.salePrice)}</ins>
                <span class="flatsome-discount-percent">-${product.discountPercent}%</span>
              </div>
              <button type="button" onclick="selectProductForQuote('${product.name.replace(/'/g, "\\'")}')" 
                class="mt-2.5 w-full py-1.5 sm:py-2 px-2 rounded-lg bg-qd-soft hover:bg-qd-primary text-qd-primary hover:text-white font-bold text-[11px] sm:text-xs transition">
                Đặt mua / Báo giá
              </button>
            </div>
          </div>
        `;

        grid.appendChild(card);
        const imgEl = card.querySelector("img");
        attachDriveImageFallback(imgEl, product.driveId);
      });
    }

    function filterByCategory(categorySlug) {
      document.querySelectorAll(".cat-filter-btn").forEach(btn => {
        if (btn.dataset.catBtn === categorySlug) {
          btn.className = "cat-filter-btn px-3.5 py-2 rounded-lg text-xs sm:text-sm font-bold bg-qd-primary text-white shadow-sm transition";

        }
      });
      renderProducts(categorySlug, "");
    }

    /**
     * Logic tự động trượt 2 Banner cùng kích thước
     */
    let currentSlide = 0;
    const totalSlides = 2;
    let slideTimer = null;

    function goToSlide(index) {
      currentSlide = (index + totalSlides) % totalSlides;
      const slides = document.querySelectorAll(".banner-slide");
      const dots = document.querySelectorAll(".slide-dot");

      slides.forEach((slide, idx) => {
        if (idx === currentSlide) {
          slide.classList.remove("opacity-0", "invisible", "z-0");
          slide.classList.add("opacity-100", "visible", "z-10");
        } else {
          slide.classList.remove("opacity-100", "visible", "z-10");
          slide.classList.add("opacity-0", "invisible", "z-0");
        }
      });

      dots.forEach((dot, idx) => {
        if (idx === currentSlide) {
          dot.className = "slide-dot w-6 h-2 rounded-full bg-amber-400 transition-all";
        } else {
          dot.className = "slide-dot w-2.5 h-2 rounded-full bg-white/60 transition-all";
        }
      });
    }

    function startBannerAutoSlide() {
      if (slideTimer) clearInterval(slideTimer);
      slideTimer = setInterval(() => {
        goToSlide(currentSlide + 1);
      }, 4500);
    }
